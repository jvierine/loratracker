cmake_minimum_required(VERSION 3.16)
project(stratos_decoder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
# Conda can supply libc++ on macOS even when CMake uses /usr/bin/c++.
# Keep installed replay tools runnable alongside the Rust HDF5 binaries.
if(APPLE AND DEFINED ENV{HDF5_DIR})
  set(CMAKE_BUILD_RPATH "$ENV{HDF5_DIR}/lib")
  set(CMAKE_INSTALL_RPATH "$ENV{HDF5_DIR}/lib")
endif()
add_library(kiss_fft STATIC vendor/lora/kiss_fft.c)
target_include_directories(kiss_fft PUBLIC vendor/lora)
add_executable(lora_rx vendor/lora/lora_rx.cpp)
target_compile_definitions(lora_rx PRIVATE STATION_PIPE_ONLY)
target_link_libraries(lora_rx PRIVATE kiss_fft Threads::Threads)
add_executable(lora_tx_gen vendor/lora/lora_tx_gen.cpp)
foreach(target lora_rx lora_tx_gen kiss_fft)
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /O2 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -O3)
  endif()
endforeach()
install(TARGETS lora_rx lora_tx_gen RUNTIME DESTINATION bin)
